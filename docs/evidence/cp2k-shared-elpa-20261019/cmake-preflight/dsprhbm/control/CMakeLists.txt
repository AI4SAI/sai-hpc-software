cmake_minimum_required(VERSION 3.22)
project(elpa_consumer LANGUAGES Fortran)
find_package(PkgConfig REQUIRED)
pkg_search_module(CP2K_ELPA REQUIRED IMPORTED_TARGET GLOBAL elpa_openmp)
add_executable(consumer /control/cp2k_elpa_abi.f90)
target_include_directories(consumer PRIVATE ${CP2K_ELPA_INCLUDE_DIRS} ${CP2K_ELPA_INCLUDE_DIRS}/modules)
target_compile_options(consumer PRIVATE -fopenmp)
target_link_options(consumer PRIVATE -fopenmp)
target_link_libraries(consumer PRIVATE ${CP2K_ELPA_LINK_LIBRARIES})
