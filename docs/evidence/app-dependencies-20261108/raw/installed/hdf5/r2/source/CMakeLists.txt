cmake_minimum_required(VERSION 3.20)
project(installed_hdf5_consumer LANGUAGES C Fortran)
find_package(MPI REQUIRED COMPONENTS C Fortran)
find_package(ZLIB REQUIRED)
include("${INSTALLED_HDF5_PREFIX}/lib/cmake/qe_szip/qe_szipConfig.cmake")
find_package(HDF5 1.14.6 EXACT CONFIG REQUIRED COMPONENTS C Fortran static
             PATHS "${INSTALLED_HDF5_PREFIX}/cmake" NO_DEFAULT_PATH)
if(NOT HDF5_ENABLE_PARALLEL OR NOT HDF5_BUILD_FORTRAN OR HDF5_BUILD_SHARED_LIBS)
  message(FATAL_ERROR "Unexpected HDF5 capabilities")
endif()
foreach(target IN ITEMS hdf5-static hdf5_f90cstub-static hdf5_fortran-static)
  get_target_property(location ${target} IMPORTED_LOCATION_RELEASE)
  if(NOT location MATCHES "^${INSTALLED_HDF5_PREFIX}/lib/")
    message(FATAL_ERROR "HDF5 imported target escaped physical prefix: ${location}")
  endif()
  file(APPEND "${CMAKE_BINARY_DIR}/provider-targets.txt" "${target}=${location}\n")
endforeach()
add_executable(consumer consumer.f90)
target_link_libraries(consumer PRIVATE hdf5_fortran-static MPI::MPI_Fortran)
